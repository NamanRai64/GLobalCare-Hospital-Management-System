-- Lookup table for Units (True 5NF instead of ENUM)
CREATE TABLE UnitOfMeasure (
    uom_id INT PRIMARY KEY,
    unit_name VARCHAR(50) NOT NULL, -- e.g., 'Millimeters of Mercury'
    unit_symbol VARCHAR(10) NOT NULL -- e.g., 'mmHg', 'bpm', '°C', 'mg/dL'
);

-- The central event table
CREATE TABLE Encounter (
    encounter_id BIGINT PRIMARY KEY,
    patient_id BIGINT NOT NULL,
    doctor_id BIGINT NOT NULL,
    branch_id BIGINT NOT NULL,
    encounter_type VARCHAR(50), -- 'Inpatient', 'Outpatient', 'ER'
    start_time DATETIME NOT NULL,
    end_time DATETIME,
    reason_for_visit TEXT,
    FOREIGN KEY (patient_id) REFERENCES Patient(patient_id),
    FOREIGN KEY (doctor_id) REFERENCES Doctor(employee_id),
    FOREIGN KEY (branch_id) REFERENCES BranchHospital(branch_id)
);

CREATE TABLE MedicalRecord (
    record_id BIGINT PRIMARY KEY,
    patient_id BIGINT,
    created_date DATE,
    FOREIGN KEY (patient_id) REFERENCES Patient(patient_id)
);

CREATE TABLE Diagnosis (
    diagnosis_id BIGINT PRIMARY KEY,
    encounter_id BIGINT NOT NULL,
    icd_code VARCHAR(10), -- International Classification of Diseases code
    diagnosis_details TEXT,
    is_provisional BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (encounter_id) REFERENCES Encounter(encounter_id)
);

CREATE TABLE Prescription (
    prescription_id BIGINT PRIMARY KEY,
    encounter_id BIGINT NOT NULL,
    doctor_id BIGINT NOT NULL,
    medicine_id BIGINT NOT NULL,
    dosage VARCHAR(100),
    frequency VARCHAR(100),
    duration VARCHAR(100),
    prescribed_date DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (encounter_id) REFERENCES Encounter(encounter_id),
    FOREIGN KEY (doctor_id) REFERENCES Doctor(employee_id),
    FOREIGN KEY (medicine_id) REFERENCES Medicine(medicine_id)
);

CREATE TABLE LabOrder (
    lab_order_id BIGINT PRIMARY KEY,
    encounter_id BIGINT NOT NULL,
    order_date DATE,
    priority VARCHAR(20), -- 'Routine' or 'Stat'
    FOREIGN KEY (encounter_id) REFERENCES Encounter(encounter_id)
);

CREATE TABLE LabResult (
    result_id BIGINT PRIMARY KEY,
    lab_order_id BIGINT NOT NULL,
    result_value NUMERIC(10, 3), 
    uom_id INT,
    result_details TEXT,
    is_abnormal BOOLEAN, -- Flag for the UI
    FOREIGN KEY (lab_order_id) REFERENCES LabOrder(lab_order_id),
    FOREIGN KEY (uom_id) REFERENCES UnitOfMeasure(uom_id)
);

CREATE TABLE ProcedureTable (
    procedure_id BIGINT PRIMARY KEY,
    record_id BIGINT,
    procedure_date DATE,
    description TEXT,
    FOREIGN KEY (record_id) REFERENCES MedicalRecord(record_id)
);

CREATE TABLE Surgery (
    surgery_id BIGINT PRIMARY KEY,
    record_id BIGINT,
    surgeon_id BIGINT,
    surgery_date DATE,
    FOREIGN KEY (record_id) REFERENCES MedicalRecord(record_id),
    FOREIGN KEY (surgeon_id) REFERENCES Doctor(employee_id)
);

-- 1. Equipment Categories (e.g., Radiology, Life Support)
CREATE TABLE EquipmentCategory (
    category_id BIGINT PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL
);

-- 2. Technicians' Skills (Binary Relationship: Tech + Category)
-- 5NF: Only store the atomic fact of capability
CREATE TABLE TechnicianSkill (
    technician_id BIGINT,
    category_id BIGINT,
    certification_level VARCHAR(50),
    PRIMARY KEY (technician_id, category_id),
    FOREIGN KEY (technician_id) REFERENCES Technician(employee_id),
    FOREIGN KEY (category_id) REFERENCES EquipmentCategory(category_id)
);

-- 3. Technicians' Departmental Access (Binary Relationship: Tech + Dept)
-- 5NF: Only store the atomic fact of location authority
CREATE TABLE TechnicianDepartmentAssignment (
    technician_id BIGINT,
    department_id BIGINT,
    assignment_date DATE,
    PRIMARY KEY (technician_id, department_id),
    FOREIGN KEY (technician_id) REFERENCES Technician(employee_id),
    FOREIGN KEY (department_id) REFERENCES Department(department_id)
);

-- 4. Equipment Inventory
-- 5NF: Atomic fact of where equipment is and what type it is
CREATE TABLE MedicalEquipment (
    equipment_id BIGINT PRIMARY KEY,
    department_id BIGINT,
    category_id BIGINT,
    equipment_name VARCHAR(255) NOT NULL,
    model_number VARCHAR(100),
    serial_number VARCHAR(100) UNIQUE,
    purchase_date DATE,
    warranty_expiry DATE,
    status VARCHAR(50), 
    FOREIGN KEY (department_id) REFERENCES Department(department_id),
    FOREIGN KEY (category_id) REFERENCES EquipmentCategory(category_id)
);

-- 5. Maintenance Log
-- This records the ACTUAL occurrence. We no longer need a separate 
-- "Assignment" table because the assignment is derived from the tables above.
CREATE TABLE MaintenanceLog (
    maintenance_id BIGINT PRIMARY KEY,
    equipment_id BIGINT,
    technician_id BIGINT,
    maintenance_date DATETIME,
    maintenance_type VARCHAR(50), 
    description TEXT,
    cost DECIMAL(15, 2),
    FOREIGN KEY (equipment_id) REFERENCES MedicalEquipment(equipment_id),
    FOREIGN KEY (technician_id) REFERENCES Technician(employee_id)
);
